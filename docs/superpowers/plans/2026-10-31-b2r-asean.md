# B2R ASEAN Intelligence Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship static-first Next.js app grouped by 11 ASEAN countries with Regulation / Market / Competitors tabs, live no-key APIs, and BYOK AI enhance.

**Architecture:** Next.js App Router SSG + `data/countries/*.json` static base; server components fetch World Bank, Frankfurter, countries.dev, GDELT with 24h revalidate and per-tab fallback; `/api/enhance` proxies BYOK key without storage.

**Tech Stack:** Next.js 15 App Router + TypeScript strict, Tailwind CSS, pnpm, vitest, zod (validation only), next/font, next/image AVIF.

## Global Constraints

- App Router only, no `pages/` directory.
- Tailwind only for styling, no component library, no chart/map library (CSS bars only).
- `revalidate = 86400` on all live server fetches, 5s timeout, 1 retry, static fallback required.
- Every `regulation[]` entry must have `sourceUrl + sourceName + lastVerified` (YYYY-MM-DD).
- BYOK key: in-memory only, `type=password` input, never logged, never committed, 400 if missing.
- Routes: `/`, `/countries/[code]`, `/compare`, `/api/enhance`.
- Country files: `data/countries/{CODE}.json` uppercase (`TH.json`), 11 codes: BN KH ID LA MY MM PH SG TH VN TL.
- Performance budget: <150KB first-load JS, Lighthouse >90, `next/image` + `next/font` required.
- Package manager: pnpm.
- Do NOT use REST Countries v3.1 (deprecated) — use countries.dev.

// lib/countries.ts
import fs from "node:fs";
import path from "node:path";
import type { CountryData } from "./types";

const DIR = path.join(process.cwd(), "data", "countries");

export function listCountries(): { code: string; name: string }[] {
  return fs
    .readdirSync(DIR)
    .filter((f) => f.endsWith(".json"))
    .map((f) => {
      const j = JSON.parse(fs.readFileSync(path.join(DIR, f), "utf8")) as CountryData;
      return { code: j.code, name: j.name };
    })
    .sort((a, b) => a.code.localeCompare(b.code));
}

export function getCountry(code: string): CountryData {
  const file = path.join(DIR, `${code.toUpperCase()}.json`);
  return JSON.parse(fs.readFileSync(file, "utf8")) as CountryData;
}

// lib/sources.ts
export const REVALIDATE = 86400;

async function withTimeout(url: string, ms = 5000): Promise<Response> {
  const ctl = new AbortController();
  const t = setTimeout(() => ctl.abort(), ms);
  try {
    return await fetch(url, { signal: ctl.signal, next: { revalidate: REVALIDATE } });
  } finally {
    clearTimeout(t);
  }
}

export async function fetchWorldBank(code: string, indicators: string[]) {
  try {
    const out: Record<string, number | null> = {};
    for (const ind of indicators.slice(0, 5)) {
      const res = await withTimeout(`https://api.worldbank.org/v2/country/${code}/indicator/${ind}?format=json&per_page=2`);
      const j = await res.json();
      const val = Array.isArray(j) && j[1]?.[0]?.value != null ? Number(j[1][0].value) : null;
      out[ind] = val;
    }
    return { ok: true, data: out, updatedAt: new Date().toISOString() };
  } catch {
    return { ok: false, data: {} as Record<string, number | null>, updatedAt: new Date().toISOString() };
  }
}

export async function fetchFx() {
  try {
    const res = await withTimeout("https://api.frankfurter.app/latest?from=USD");
    const j = await res.json();
    return { ok: true, data: j.rates as Record<string, number>, updatedAt: j.date as string };
  } catch {
    return { ok: false, data: {} as Record<string, number>, updatedAt: "" };
  }
}

export async function fetchGdelt(query: string) {
  try {
    const res = await withTimeout(
      `https://api.gdeltproject.org/api/v2/doc/doc?query=${encodeURIComponent(query)}&mode=artlist&maxrecords=5&format=json`
    );
    const j = await res.json();
    const items = (j.articles ?? []).map((a: { title: string; url: string; domain: string; seendate: string }) => ({
      title: a.title, url: a.url, domain: a.domain, seendate: a.seendate,
    }));
    return { ok: true, data: items, updatedAt: new Date().toISOString() };
  } catch {
    return { ok: false, data: [] as { title: string; url: string; domain: string; seendate: string }[], updatedAt: "" };
  }
}

export async function fetchBasics(code: string) {
  try {
    const res = await withTimeout(`https://countries.dev/alpha/${code}`);
    const j = await res.json();
    return { ok: true, data: j, updatedAt: new Date().toISOString() };
  } catch {
    return { ok: false, data: null, updatedAt: "" };
  }
}

// app/api/enhance/route.ts
import { NextResponse } from "next/server";
import { getCountry } from "@/lib/countries";

export async function POST(req: Request) {
  const body = await req.json().catch(() => null) as { code?: string; industry?: string; apiKey?: string } | null;
  if (!body?.code || !body?.industry || !body?.apiKey) {
    return NextResponse.json({ error: "code, industry and apiKey are required (BYOK, never stored)." }, { status: 400 });
  }
  const country = getCountry(body.code);

"use client";
import { useState } from "react";

export default function EnhanceBox({ code, industry }: { code: string; industry: string }) {
  const [key, setKey] = useState("");
  const [out, setOut] = useState("");
  const [busy, setBusy] = useState(false);
  async function run() {
    setBusy(true);
    try {
      const res = await fetch("/api/enhance", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({ code, industry, apiKey: key }),
      });
      const j = await res.json();
      setOut(j.markdown ?? j.error ?? "Failed");
    } finally {
      setBusy(false);
      setKey("");
    }
  }
  return (
    <div className="border rounded p-4 space-y-2">
      <div className="font-medium text-sm">AI enhance (BYOK — key never stored)</div>
      <input type="password" value={key} onChange={(e) => setKey(e.target.value)} placeholder="Paste key to enable" className="border rounded px-2 py-1 w-full text-sm" />
      <button disabled={!key || busy} onClick={run} className="text-sm border rounded px-3 py-1 disabled:opacity-50">
        {busy ? "Working…" : "Generate angle"}
      </button>
      {out && <pre className="text-xs whitespace-pre-wrap">{out}</pre>}
    </div>
  );
}
